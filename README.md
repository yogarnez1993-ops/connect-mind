-- EXTENSIONES
create extension if not exists "pg_trgm"; create extension if not exists "pgcrypto";

-- PACIENTES
create table usuarios_pacientes (id uuid primary key default gen_random_uuid(), nombre text not null check (char_length(nombre)>=3 and char_length(nombre)<=100), dni text unique not null check (dni ~ '^[0-9]{8}$'), celular text unique not null check (celular ~ '^9[0-9]{8}$'), email text unique not null check (email ~* '^[^@]+@[^@]+\.[^@]+$'), password_hash text not null, categoria text check (categoria in ('Social','Medium','Premium')), estado text not null default 'prueba_pendiente' check (estado in ('prueba_pendiente','paquete_prueba_aprobado','activo','bloqueado','finalizado')), problemas_elegidos jsonb default '[]'::jsonb, acepto_politicas boolean default true not null, ip_registro text, user_agent text, fecha_registro timestamptz default now() not null);
create index idx_pacientes_dni on usuarios_pacientes(dni); create index idx_pacientes_celular on usuarios_pacientes(celular); create index idx_pacientes_email on usuarios_pacientes(email); create index idx_pacientes_estado on usuarios_pacientes(estado); create index idx_pacientes_categoria on usuarios_pacientes(categoria); create index idx_pacientes_fecha on usuarios_pacientes(fecha_registro desc);
alter table usuarios_pacientes enable row level security; create policy p_pacientes_self on usuarios_pacientes for all using (auth.uid()::text = id::text);

-- PSICOLOGOS
create table psicologos (id uuid primary key default gen_random_uuid(), nombre text not null, dni text not null check (dni ~ '^[0-9]{8}$'), cnp text unique not null check (cnp ~ '^[0-9]{4,6}$'), celular text not null, email text unique not null, foto_url text, especialidades jsonb default '[]'::jsonb, categoria text not null check (categoria in ('Social','Medium','Premium','Fundador')), tipo_pago text default 'normal_55' check (tipo_pago in ('normal_55','fundador_70')), ranking_cierres int default 0 not null, estado text default 'activo' check (estado in ('activo','inactivo','bloqueado')), agenda_disponible jsonb default '[]'::jsonb, rating numeric(2,1) default 5.0 check (rating>=0 and rating<=5));
create index idx_psicologos_categoria_estado on psicologos(categoria, estado); create index idx_psicologos_ranking_desc on psicologos(ranking_cierres desc); create index idx_psicologos_cnp on psicologos(cnp); create index idx_psicologos_categoria on psicologos(categoria);
alter table psicologos enable row level security;

-- PROBLEMAS
create table problemas_psicologicos (id uuid primary key default gen_random_uuid(), nombre text unique not null, descripcion_corta text, icono text default 'Heart', activo boolean default true not null, orden int default 0 not null);
create index idx_problemas_activo_orden on problemas_psicologicos(activo, orden); create index idx_problemas_trgm on problemas_psicologicos using gin (nombre gin_trgm_ops);

-- PAQUETES
create table paquetes (id uuid primary key default gen_random_uuid(), nombre text not null, sesiones int not null check (sesiones>0), categoria text not null check (categoria in ('Social','Medium','Premium')), precio_inicio numeric(10,2) not null, precio_fin numeric(10,2) not null, vacantes_social int default 100, activo boolean default true, descripcion text, beneficios jsonb default '[]'::jsonb);

-- SESION PRUEBA
create table sesiones_prueba (id uuid primary key default gen_random_uuid(), paciente_id uuid not null references usuarios_pacientes(id) on delete cascade, categoria text not null, precio numeric(10,2) not null, metodo_pago text not null check (metodo_pago in ('Yape','Plin','Transferencia','Efectivo')), voucher_url text not null, estado_voucher text default 'pendiente' check (estado_voucher in ('pendiente','aprobado','rechazado')), fecha_pago timestamptz default now(), notas_admin text, ip_subida text);
create index idx_ses_prueba_estado_fecha on sesiones_prueba(estado_voucher, fecha_pago desc); create index idx_ses_prueba_paciente on sesiones_prueba(paciente_id);

-- PAGOS PAQUETES
create table pagos_paquetes (id uuid primary key default gen_random_uuid(), paciente_id uuid not null references usuarios_pacientes(id) on delete cascade, paquete_id uuid not null references paquetes(id), psicologo_asignado_id uuid references psicologos(id), monto_pagado numeric(10,2) not null, voucher_url text not null, estado_voucher text default 'pendiente' check (estado_voucher in ('pendiente','aprobado','rechazado')), estado text default 'pendiente_aprobacion' check (estado in ('pendiente_aprobacion','activo','finalizado','cancelado')), sesiones_totales int not null, sesiones_hechas int default 0 check (sesiones_hechas>=0), fecha_inicio timestamptz, fecha_fin timestamptz, tipo_terapia text default 'Individual' check (tipo_terapia in ('Individual','Pareja','Familiar')));
create index idx_pagos_paciente on pagos_paquetes(paciente_id); create index idx_pagos_psicologo on pagos_paquetes(psicologo_asignado_id); create index idx_pagos_estado on pagos_paquetes(estado);

-- AGENDA PARTICIONADA MENSUAL MILLONES
create table agenda_citas (id uuid primary key default gen_random_uuid(), paciente_id uuid not null references usuarios_pacientes(id) on delete cascade, psicologo_id uuid not null references psicologos(id) on delete cascade, fecha_hora timestamptz not null, estado text default 'agendada' check (estado in ('agendada','realizada','cancelada','pendiente_agendar_siguiente')), tipo text default 'Individual' check (tipo in ('Individual','Pareja','Familiar')), color text default 'PLATA' check (color in ('VERDE','DIAMANTE','ORO','PLATA')), duracion int default 50, link_video text, recordatorio_enviado boolean default false) partition by range (fecha_hora);
create table agenda_citas_2026_01 partition of agenda_citas for values from ('2026-01-01') to ('2026-02-01'); create table agenda_citas_2026_02 partition of agenda_citas for values from ('2026-02-01') to ('2026-03-01'); create table agenda_citas_2026_03 partition of agenda_citas for values from ('2026-03-01') to ('2026-04-01'); create table agenda_citas_default partition of agenda_citas default;
create index idx_agenda_psicologo_fecha_estado on agenda_citas(psicologo_id, fecha_hora, estado); create index idx_agenda_paciente_fecha on agenda_citas(paciente_id, fecha_hora desc); create index idx_agenda_fecha_estado on agenda_citas(fecha_hora, estado); create index idx_agenda_pendiente on agenda_citas(estado, fecha_hora) where estado='pendiente_agendar_siguiente';

-- BILLETERA PARTICIONADA
create table billetera_psicologo (id uuid primary key default gen_random_uuid(), psicologo_id uuid not null references psicologos(id) on delete cascade, tipo text not null check (tipo in ('adelanto_20','pago_sesion_31.5','bono_fundador')), monto numeric(10,2) not null, descripcion text, cita_id uuid references agenda_citas(id), fecha_generacion timestamptz default now(), estado text default 'pendiente' check (estado in ('pendiente','pagado')), fecha_pago_sabado date not null) partition by range (fecha_generacion);
create table billetera_2026_01 partition of billetera_psicologo for values from ('2026-01-01') to ('2026-02-01'); create table billetera_default partition of billetera_psicologo default;
create index idx_billetera_psicologo on billetera_psicologo(psicologo_id, fecha_generacion desc); create index idx_billetera_estado_fecha on billetera_psicologo(estado, fecha_pago_sabado);

-- RESTO
create table cortes_semanales (id uuid primary key default gen_random_uuid(), fecha_corte_viernes date not null, fecha_pago_sabado date not null, total_recaudado numeric(12,2) default 0, ganancia_empresa_45 numeric(12,2) default 0, total_pagar_psicologos numeric(12,2) default 0, estado text default 'abierto' check (estado in ('abierto','cerrado','pagado')));
create table notas_clinicas (id uuid primary key default gen_random_uuid(), paciente_id uuid not null references usuarios_pacientes(id), psicologo_id uuid not null references psicologos(id), cita_id uuid unique not null references agenda_citas(id), contenido_encriptado text not null, iv text not null, version int default 1, fecha_creacion timestamptz default now(), para_susalud boolean default false, visible_paciente boolean default false);
create table videollamadas (id uuid primary key default gen_random_uuid(), cita_id uuid unique not null references agenda_citas(id), sala_id text unique not null, creador_psicologo_id uuid references psicologos(id), participantes jsonb default '[]'::jsonb, estado text default 'sala_espera' check (estado in ('sala_espera','en_curso','finalizada')), inicio timestamptz, fin timestamptz, grabacion_permitida boolean default false, consentimiento_grabacion boolean default false);
create table comprobantes_sunat (id uuid primary key default gen_random_uuid(), paciente_id uuid not null references usuarios_pacientes(id), pago_id uuid, tipo text not null check (tipo in ('boleta','factura')), serie text not null default 'B001', correlativo int not null, monto numeric(10,2) not null, igv numeric(10,2) default 0, total numeric(10,2) not null, pdf_url text, fecha_emision timestamptz default now(), estado text default 'emitido');
create index idx_comprobantes_serie_corr on comprobantes_sunat(serie, correlativo desc); create index idx_comprobantes_paciente on comprobantes_sunat(paciente_id);
create table recibos_honorarios (id uuid primary key default gen_random_uuid(), psicologo_id uuid not null references psicologos(id), monto numeric(10,2) not null, mes text not null, archivo_url text not null, estado text default 'pendiente' check (estado in ('pendiente','aprobado','rechazado')), fecha_subida timestamptz default now(), numero_recibo text, serie text default 'E001');
create table configuracion_app (clave text primary key, valor text not null, descripcion text, editable boolean default true, version int default 1);
create table notificaciones (id uuid primary key default gen_random_uuid(), usuario_id uuid, psicologo_id uuid, tipo text not null, canal text not null check (canal in ('push','whatsapp','email')), enviada boolean default false, fecha timestamptz default now(), intentos int default 0, provider_response jsonb);
create table logs_admin (id uuid primary key default gen_random_uuid(), admin_id uuid, accion text not null, tabla_afectada text, registro_id uuid, fecha timestamptz default now(), ip text, detalles jsonb);
create table logs_acceso_clinico (id uuid primary key default gen_random_uuid(), admin_id uuid, psicologo_id uuid, nota_id uuid references notas_clinicas(id), fecha timestamptz default now(), ip text, motivo text not null);
create table reclamaciones (id uuid primary key default gen_random_uuid(), paciente_id uuid references usuarios_pacientes(id), nombre text not null, reclamo text not null, detalle text, fecha timestamptz default now(), estado text default 'pendiente' check (estado in ('pendiente','atendido')));
